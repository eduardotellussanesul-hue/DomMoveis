npm install -g pm2

pm2 start "npm run dev" --name dommoveis-api

pm2 save
pm2 startup

# ver processos
pm2 status

pm2 restart dommoveis-api

# parar sem remover
pm2 stop dommoveis-api

# subir novamente (se estava parado)
pm2 start dommoveis-api

# derrubar e remover da lista
pm2 delete dommoveis-api

# ver logs em tempo real
